function castOps(value: f32, ptr: *mut u8): void {
  let bits: u32 = @bitCast<u32>(value);
  let addr: u64 = @integerFromPointer(ptr);
  let newPtr: *mut u8 = @pointerFromInteger<*mut u8>(addr);
  let castPtr: *mut i32 = @pointerCast<*mut i32>(newPtr);

  bits;
  castPtr;
}

function failFast(enabled: boolean): void {
  if (enabled) {
    @panic("boom");
  } else {
    @trap();
  }

  @unreachable();
}

function compileGuard(): void {
  @compileError("this should fail in compiler checks");
}
