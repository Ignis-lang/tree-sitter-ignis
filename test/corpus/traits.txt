record Buffer {
  public data: string;

  clone(&self): Buffer {
    return Buffer { data: self.data };

}

@extension(i32)
function doubled(value: i32): i32 {
  return value * 2;
}

@extension(i32, mut)
function addSelf(value: i32): i32 {
  return value + value;
}

function main(): i32 {
  let mut x: i32 = 21;
  let y: i32 = x.doubled();
  return y.addSelf();
}
